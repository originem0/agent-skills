```python
import sys, threading, time

def work(n=10**7):
    x = 0
    for i in range(n):
        x += i

print(sys.version)

t = time.perf_counter()
for _ in range(4):
    work()
print("单线程跑 4 份：", round(time.perf_counter() - t, 2), "s")

t = time.perf_counter()
threads = [threading.Thread(target=work) for _ in range(4)]
for th in threads: th.start()
for th in threads: th.join()
print("4 线程各跑 1 份：", round(time.perf_counter() - t, 2), "s")
```
